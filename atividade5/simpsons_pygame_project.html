🏠 Cenário Os Simpsons em Pygame

Simulação interativa da atividade prática de Práticas de Programação.

Espaço: Alternar Cor do Céu
M: Som "D'oh!"
Clique: Mover Homer
Nuvem Animada (XP Extra Active: 100px/s)

🐍 Código Python / Pygame Correspondente (Para Entrega) 

from pygame import *

init()
screen = display.set_mode((1280, 720))
display.set_caption("Casinha dos Simpsons - Trabalho Final")
running = True
clock = time.Clock()

## 1. Carregamento de Recursos (Fontes, Imagens e Áudio)
try:
    fonte = font.Font("Simpsonfont.ttf", 65)
except:
    fonte = font.SysFont("Comic Sans MS", 65, bold=True)

# Carrega e redimensiona o Homer
homer_img = image.load("homer.png")
homer_img = transform.scale(homer_img, (140, 220))

# Carrega e redimensiona o Arbusto / Moita
moita_img = image.load("moita.png")
moita_img = transform.scale(moita_img, (220, 150))

# Carregamento e Execução do Áudio de Fundo
mixer.music.load("simpsons_theme.mp3")
mixer.music.play(-1) # -1 faz a música rodar em loop infinito

# Efeito Sonoro Específico (Sfx)
doh_sfx = mixer.Sound("doh.mp3")

## 2. Definição de Variáveis de Controle e Posições
pos_nuvem_x = 300  # Posição X da Nuvem (Para a animação do EXTRA)
velocidade_nuvem = 120 # Pixels por segundo
background_color = "#87CEEB" # Azul Céu dos Simpsons
texto_titulo = "Simpsons"

## 3. Loop Principal do Jogo
while running:
    # Garante a taxa constante de 60 Quadros Por Segundo (FPS)
    dt = clock.tick(60) / 1000.0

    # Trata todos os Eventos
    for ev in event.get():
        if ev.type == QUIT:
            running = False
        
        # Interação via Teclado
        if ev.type == KEYDOWN:
            if ev.key == K_SPACE:
                # Altera a cor do céu ao pressionar espaço
                background_color = "#F5B240" if background_color == "#87CEEB" else "#87CEEB"
            elif ev.key == K_m:
                doh_sfx.play()

    ## 4. Atualização da Lógica (EXTRA - Nuvem Infinita)
    pos_nuvem_x += velocidade_nuvem * dt
    # Quando a nuvem sai totalmente da tela à direita (> 1280), teleporta para a esquerda (-250)
    if pos_nuvem_x > 1280:
        pos_nuvem_x = -250

    ## 5. Renderização / Desenho do Cenário (Primitivas e Elementos)
    
    # 5.1 Fundo (Céu)
    screen.fill(background_color)

    # 5.2 Gramado (Retângulo Verde na base)
    draw.rect(screen, "#38A169", (0, 520, 1280, 200))

    # 5.3 Sol Radiante (Círculo e Linhas para os Raios)
    centro_sol = (180, 150)
    draw.circle(screen, "#FFD700", centro_sol, 55) # Centro do Sol
    
    # Desenho dos 8 Raios do Sol
    raios = [
        ((180, 75), (180, 45)),   # Cima
        ((180, 225), (180, 255)), # Baixo
        ((105, 150), (75, 150)),  # Esquerda
        ((255, 150), (285, 150)), # Direita
        ((127, 97), (105, 75)),   # Diag Sup Esq
        ((233, 97), (255, 75)),   # Diag Sup Dir
        ((127, 203), (105, 225)), # Diag Inf Esq
        ((233, 203), (255, 225))  # Diag Inf Dir
    ]
    for inicio, fim in raios:
        draw.line(screen, "#FFD700", inicio, fim, 5)

    # 5.4 Nuvem Animada (Agrupamento de Círculos)
    y_nuvem = 90
    draw.circle(screen, "#FFFFFF", (int(pos_nuvem_x), y_nuvem), 45)
    draw.circle(screen, "#FFFFFF", (int(pos_nuvem_x) + 40, y_nuvem - 15), 50)
    draw.circle(screen, "#FFFFFF", (int(pos_nuvem_x) + 90, y_nuvem - 10), 45)
    draw.circle(screen, "#FFFFFF", (int(pos_nuvem_x) + 130, y_nuvem), 40)

    # 5.5 Texto Estilizado ("Simpsons") entre a nuvem e a árvore
    texto_surface = fonte.render(texto_titulo, True, "#FFD700")
    # Pequena sombra para destacá-lo
    sombra_surface = fonte.render(texto_titulo, True, "#000000")
    screen.blit(sombra_surface, (723, 163))
    screen.blit(texto_surface, (720, 160))

    # 5.6 Casa dos Simpsons (Primitivas com as cores oficiais)
    # Telhado Triangular
    draw.polygon(screen, "#BB5C38", [(250, 480), (550, 480), (400, 280)])
    # Parede Principal Rosa/Bege
    draw.rect(screen, "#E19B7C", (280, 480, 240, 170))
    # Porta Marrom
    draw.rect(screen, "#8B4513", (410, 530, 50, 120))
    # Maçaneta Dourada
    draw.circle(screen, "#FFD700", (420, 595), 4)
    # Janela Azul Escuro
    draw.rect(screen, "#1A2B56", (310, 530, 60, 60))

    # 5.7 Árvore do Cenário
    # Tronco Marrom
    draw.rect(screen, "#654321", (980, 420, 60, 230))
    # Copa Verde
    draw.circle(screen, "#2E8B57", (1010, 380), 100)

    # 5.8 Homer no Arbusto (Entre a Casa e a Árvore)
    # Homer é desenhado PRIMEIRO (Fica atrás)
    screen.blit(homer_img, (710, 440))
    # Arbusto / Moita é desenhada DEPOIS (Cobre o Homer parcialmente)
    screen.blit(moita_img, (670, 510))

    ## 6. Atualiza o Display
    display.update()

quit()
